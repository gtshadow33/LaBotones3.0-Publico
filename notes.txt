mvn  javafx:run

//linux
jpackage \
  --type app-image \
  --name LaBotones \
  --input target \
  --main-jar labotones-1.0-SNAPSHOT.jar \
  --main-class com.labotones.App \
  --dest dist \
  --module-path /usr/share/openjfx/lib \
  --add-modules javafx.controls,javafx.fxml

  --name LaBotones ^
  --input target ^
  --main-jar labotones-1.0-SNAPSHOT.jar ^

  --dest dist ^
  --module-path "C:\javafx\lib" ^
